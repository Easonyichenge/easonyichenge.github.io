            }

            

            function renderHistoryForDate(date) {

                historyList.innerHTML = '';

                const records = attendanceRecords[date] || {};

                const studentIds = Object.keys(records).map(id => parseInt(id));

                if (studentIds.length === 0) {

                    historyList.innerHTML = `<p class="text-gray-500 text-center py-4">此日期無紀錄。</p>`; return;

                }

                students.filter(s => studentIds.includes(s.id)).forEach(s => {

                    const status = records[s.id];

                    const statusText = { present: '到課', absent: '缺席', late: '遲到' }[status] || '未記錄';

                    const statusColor = { present: 'text-green-600', absent: 'text-red-600', late: 'text-yellow-600' }[status] || 'text-gray-500';

                    const item = document.createElement('div');

                    item.className = 'flex items-center justify-between p-4 bg-gray-50 rounded-xl';

                    item.innerHTML = `<p>${s.name}</p><p class="font-semibold ${statusColor}">${statusText}</p>`;

                    historyList.appendChild(item);

                });

            }

            // --- Modal 邏輯 ---

            function openModal(modalEl) {

                const backdrop = document.createElement('div');

                backdrop.className = 'modal-backdrop fixed inset-0';

                modalContainer.appendChild(backdrop);

                

                const card = document.createElement('div');

                card.className = 'apple-card relative';

                card.appendChild(modalEl);

                modalContainer.appendChild(card);

                

                modalEl.classList.remove('hidden');

                modalContainer.classList.remove('hidden');

                modalContainer.classList.add('flex', 'items-center', 'justify-center', 'p-4');


                const closeModal = () => {

                    modalContainer.innerHTML = '';

                    modalContainer.classList.add('hidden');

                    modalContainer.classList.remove('flex', 'items-center', 'justify-center', 'p-4');

                };

                

                backdrop.addEventListener('click', closeModal);

                modalContainer.querySelectorAll('.modal-close-btn').forEach(btn => btn.addEventListener('click', closeModal));

                return closeModal;

            }

            

            function showConfirm(title, onOk) {

                document.getElementById('confirm-title').textContent = title;

                const close = openModal(confirmModal);

                

                const okBtn = document.getElementById('confirm-btn-ok');

                const cancelBtn = document.getElementById('confirm-btn-cancel');

                

                const handleOk = () => { onOk(); close(); removeListeners(); };

                const handleCancel = () => { close(); removeListeners(); };


                okBtn.addEventListener('click', handleOk);

                cancelBtn.addEventListener('click', handleCancel);


                function removeListeners() {

                    okBtn.removeEventListener('click', handleOk);

                    cancelBtn.removeEventListener('click', handleCancel);

                }

            }



            // --- Gemini API ---

            async function callGeminiAPI(prompt) {

                // ... (Gemini API 呼叫邏輯與之前版本相同，為求簡潔省略)

                const apiKey = ""; const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

                const systemPrompt = "你是一位友善且經驗豐富的助教，專門協助老師處理班級事務。請根據提供的點名紀錄，用繁體中文、條列式的方式，提供一個簡潔的總結和具體的關懷建議。";

                const payload = { contents: [{ parts: [{ text: prompt }] }], systemInstruction: { parts: [{ text: systemPrompt }] }, };

                try {

                    const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });

                    if (!response.ok) throw new Error(`API 請求失敗: ${response.status}`);

                    const result = await response.json();

                    return result.candidates?.[0]?.content?.parts?.[0]?.text || "無法從 AI 獲得回應。";

                } catch (error) { console.error('Fetch Error:', error); return "無法連接至 AI 服務。"; }

            }
